  function collide(p,ball,power){let dx=ball.x-p.x,dy=ball.y-p.y,d=Math.hypot(dx,dy),min=p.r+ball.r;if(d<min&&d>0){let nx=dx/d,ny=dy/d,over=min-d;ball.x+=nx*over;ball.y+=ny*over;let rel=(p.vx-ball.vx)*nx+(p.vy-ball.vy)*ny;ball.vx+=nx*(4.5+Math.max(0,rel))*power;ball.vy+=ny*(4.5+Math.max(0,rel))*power;p.vx-=nx*1.1;p.vy-=ny*1.1;spawn(ball.x,ball.y,p.color,7);sfx(220,.018)}}
