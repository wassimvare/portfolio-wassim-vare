    function renderMarket(){
    let data=[...catalog];const q=($('#marketSearch')?.value||'').toLowerCase(),r=$('#marketRarity')?.value||'all',s=$('#marketSort')?.value||'trend';
    data=data.filter(i=>(!q||i.name.toLowerCase().includes(q))&&(r==='all'||i.rarity===r));
    data.sort((a,b)=>s==='priceAsc'?price(a)-price(b):s==='priceDesc'?price(b)-price(a):s==='scarcity'?(rarityOrder[b.rarity]-rarityOrder[a.rarity]):Math.abs(b.trend)-Math.abs(a.trend));
    $('#marketGrid').innerHTML=data.map(i=>`<article class="market-row panel" style="${styleVars(i)}"><div class="mini-art"></div><div class="market-info"><h3>${i.name}</h3><p>${i.type} • ${i.max?i.max.toLocaleString('fr-FR')+' max':'édition ouverte'} • <span class="rarity ${rarityClass(i.rarity)}">${i.rarity}</span></p><span class="trend ${i.trend>=0?'up':'down'}">${i.trend>=0?'▲':'▼'} ${Math.abs(i.trend).toFixed(1)}% / 24h</span></div><div class="market-buy"><b>${price(i).toLocaleString('fr-FR')} NC</b><button class="secondary-btn" data-buy="${i.id}">ACHETER</button></div></article>`).join('');
    $$('[data-buy]').forEach(b=>b.onclick=()=>buyItem(b.dataset.buy));$('#listingCount').textContent=(327+marketTick%19).toLocaleString('fr-FR');$('#volume24').textContent=(128400+marketTick*117).toLocaleString('fr-FR')+' NC';
  }
  ['marketSearch','marketRarity','marketSort'].forEach(id=>$('#'+id).addEventListener(id==='marketSearch'?'input':'change',renderMarket));
  function buyItem(id){const i=itemById(id),p=price(i);if(state.credits<p)return toast('Crédits insuffisants');state.credits-=p;state.inventory.push({id,serial:i.max?Math.floor(Math.random()*i.max)+1:Math.floor(Math.random()*99999)+1,wins:0,owners:Math.floor(Math.random()*4)+1});state.history.push(`Achat marché : ${i.name}`);save();toast(`${i.name} ajouté au Vault`)}
  function renderTradeInventory(){
    $('#tradeInventory').innerHTML=state.inventory.map((o,idx)=>{const i=itemById(o.id);return `<div class="trade-mini" data-trade-add="${idx}" style="${styleVars(i)}"><div class="mini-art"></div><small>${i.name}</small></div>`}).join('');
    $$('[data-trade-add]').forEach(e=>e.onclick=()=>{const idx=+e.dataset.tradeAdd;if(!offer.includes(idx)&&offer.length<4){offer.push(idx);updateTrade()}})
  }
  function generateRivalOffer(){const shuffled=[...catalog].sort(()=>Math.random()-.5);rivalOffer=shuffled.slice(0,2).map(i=>({item:i,serial:i.max?Math.ceil(Math.random()*i.max):Math.ceil(Math.random()*99999)}))}
  generateRivalOffer();
  function updateTrade(){
    const mine=offer.reduce((a,idx)=>a+price(itemById(state.inventory[idx]?.id)),0),rival=rivalOffer.reduce((a,o)=>a+price(o.item),0);$('#myOfferValue').textContent=mine.toLocaleString('fr-FR')+' NC';$('#rivalOfferValue').textContent=rival.toLocaleString('fr-FR')+' NC';
    $('#myOffer').innerHTML=offer.length?offer.map(idx=>{const i=itemById(state.inventory[idx].id);return `<button class="offer-token" data-remove="${idx}">${i.name} ×</button>`}).join(''):'Ajoute des objets depuis ton inventaire';$$('[data-remove]').forEach(b=>b.onclick=()=>{offer=offer.filter(x=>x!==+b.dataset.remove);updateTrade()});
    $('#rivalOffer').innerHTML=rivalOffer.map(o=>`<div class="rival-card" style="${styleVars(o.item)}"><div class="mini-art"></div><span>${o.item.name}<br>#${o.serial}</span></div>`).join('');
    const ratio=rival?mine/rival:0,diff=Math.abs(1-ratio),fair=mine&&diff<.14;$('#fairnessText').textContent=!mine?'—':fair?'ÉQUITABLE':ratio<1?'TU DONNES TROP PEU':'TU DONNES TROP';$('#fairnessText').style.color=fair?'var(--green)':!mine?'white':'var(--gold)';$('#fairMeter').style.width=Math.min(100,Math.max(0,(1-diff)*100))+'%';$('#acceptTrade').disabled=!fair;
  }
  $('#resetTrade').onclick=()=>{offer=[];generateRivalOffer();updateTrade()};$('#acceptTrade').onclick=()=>{const remove=[...offer].sort((a,b)=>b-a);remove.forEach(idx=>state.inventory.splice(idx,1));rivalOffer.forEach(o=>state.inventory.push({id:o.item.id,serial:o.serial,wins:0,owners:Math.floor(Math.random()*5)+2}));state.trades++;state.history.push(`Trade conclu (${rivalOffer.length} objets reçus)`);offer=[];generateRivalOffer();save();toast('Trade accepté — Vault mis à jour')};
